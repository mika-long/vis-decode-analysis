Define custom posterior predictive check function:

custom pp_check 
custom_pp_check <- function(
  fitted_model,
  fun = bayesplot::ppc_dens_overlay,
  ndraws = 100,
  transform = identity,
  ...
) {
  y_rep <- as_draws_rvars(fitted_model)$y_rep |>
    transform() |>
    as_draws_matrix()
  if (!is.null(ndraws)) {
    y_rep = y_rep |>
      merge_chains() |>
      resample_draws(ndraws = ndraws)
  }
  y <- fitted_model$draws("x_org", format = "matrix")[1, ] |>
    as.vector() |>
    transform()
  fun(y, y_rep, ...)
}
ImportantTLDR
  • centered parameterization is the most straight-forward mapping form math model to fitted model, but it creates problems for sampling (e.g., going into a funnel)
  • non-centered parameteriation makes things easier to fit; the informative prior was obtained from HighestPoint_X operator.

Centered Parameterization (informative priors)

data {
  int<lower=1> N;   // total number of observations
  int<lower=1> J;   // total number of participants 
  array[N] int<lower=1, upper=J> id; // participant ID for each observation 
  vector[N] x;      // observed values
  vector[N] x_med;  // known median values
  vector[N] x_mod;  // known mode values
}

parameters {
  // Population-level parameters 
  real mu_mod_mean;
  real<lower=0> mu_mod_sd; 
  real mu_med_mean;
  real<lower=0> mu_med_sd; 
  
  // standard deviations for x_med and x_mod
  real log_sigma_mod_mean; 
  real<lower=0> log_sigma_mod_sd; 
  real log_sigma_med_mean; 
  real<lower=0> log_sigma_med_sd;

  // Particiipant-level parameters
  vector[J] mu_mod; 
  vector<lower=0>[J] sigma_mod; 
  vector[J] mu_med; 
  vector<lower=0>[J] sigma_med; 
}

transformed parameters {
  vector<lower=0, upper=1>[N] theta; // weight 
  vector[N] inv_mse_med;
  vector[N] inv_mse_mod;  

  for (n in 1:N) {
    int j = id[n]; 
    inv_mse_med[n] = 1 / (square(mu_med[j]) + square(sigma_med[j])); 
    inv_mse_mod[n] = 1 / (square(x_mod[n] - x_med[n] + mu_mod[j]) + square(sigma_mod[j])); 

    theta[n] = inv_mse_med[n] / (inv_mse_med[n] + inv_mse_mod[n]); 
  }
}

model {
  // priors
  mu_mod_mean ~ normal(0.01, 0.02); 
  mu_mod_sd ~ normal(0.04, 0.03); 
  log_sigma_mod_mean ~ normal(-1.37, 0.10); 
  log_sigma_mod_sd ~ normal(0.34, 0.10); 

  mu_med_mean ~ normal(0, 1); 
  mu_med_sd ~ normal(0, 1); 
  log_sigma_med_mean ~ normal(0, 1); 
  log_sigma_med_sd ~ normal(0, 1); 

  for (j in 1:J){
    mu_mod[j] ~ normal(mu_mod_mean, mu_mod_sd);
    sigma_mod[j] ~ lognormal(log_sigma_mod_mean, log_sigma_mod_sd); 
    mu_med[j] ~ normal(mu_med_mean, mu_med_sd); 
    sigma_med[j] ~ lognormal(log_sigma_med_mean, log_sigma_med_sd); 
  }

  // likelihood
  for (n in 1:N) {
    int j = id[n]; 
    target += normal_lpdf(x[n] | theta[n] * (x_med[n] + mu_med[j]) + (1 - theta[n]) * (x_mod[n] + mu_mod[j]),
                                 sqrt(square(theta[n]) * square(sigma_med[j]) + square(1 - theta[n]) * square(sigma_mod[j]))); 
  }
}
generated quantities {
  vector[N] log_lik; 
  vector[N] y_rep; // posterior predictive checks
  vector[N] x_org = x;
  
  for (n in 1:N) {
    int j = id[n]; 
    log_lik[n] = normal_lpdf(x[n] | theta[n] * (x_med[n] + mu_med[j]) + (1 - theta[n]) * (x_mod[n] + mu_mod[j]),
                                 sqrt(square(theta[n]) * square(sigma_med[j]) + square(1 - theta[n]) * square(sigma_mod[j])));  

    y_rep[n] = normal_rng(theta[n] * (x_med[n] + mu_med[j]) + (1 - theta[n]) * (x_mod[n] + mu_mod[j]),
                                 sqrt(square(theta[n]) * square(sigma_med[j]) + square(1 - theta[n]) * square(sigma_mod[j])));  

  }
}

here is the corresponding math model:

\[ \begin{align*} % Likelihood x_n &\sim \mathcal{N}\!\left(\theta_n \cdot (x^{\text{med}}_n + \mu^{\text{med}}_{j[n]}) + (1 - \theta_n) \cdot (x^{\text{mod}}_n + \mu^{\text{mod}}_{j[n]}),\ \ \tilde\sigma_n\right) \\[4pt] \tilde\sigma_n &= \sqrt{\theta_n^2 \cdot (\sigma^{\text{med}}_{j[n]})^2 + (1 - \theta_n)^2 \cdot (\sigma^{\text{mod}}_{j[n]})^2} \\[4pt] % Weights (deterministic, inverse-MSE pooling) \theta_n &= \frac{\text{invMSE}^{\text{med}}_n}{\text{invMSE}^{\text{med}}_n + \text{invMSE}^{\text{mod}}_n} \\[4pt] \text{invMSE}^{\text{med}}_n &= \left[(\mu^{\text{med}}_{j[n]})^2 + (\sigma^{\text{med}}_{j[n]})^2\right]^{-1} \\[2pt] \text{invMSE}^{\text{mod}}_n &= \left[(x^{\text{mod}}_n - x^{\text{med}}_n + \mu^{\text{mod}}_{j[n]})^2 + (\sigma^{\text{mod}}_{j[n]})^2\right]^{-1} \\[6pt] % Participant-level parameters \mu^{\text{mod}}_j &\sim \mathcal{N}(\bar\mu^{\text{mod}},\ \bar\sigma^{\text{mod}}_\mu) \\ \sigma^{\text{mod}}_j &\sim \text{Lognormal}(\bar\mu^{\text{mod}}_{\log\sigma},\ \bar\sigma^{\text{mod}}_{\log\sigma}) \\ \mu^{\text{med}}_j &\sim \mathcal{N}(\bar\mu^{\text{med}},\ \bar\sigma^{\text{med}}_\mu) \\ \sigma^{\text{med}}_j &\sim \text{Lognormal}(\bar\mu^{\text{med}}_{\log\sigma},\ \bar\sigma^{\text{med}}_{\log\sigma}) \\[6pt] % Hyperpriors — mod (informative, from brms fit) \bar\mu^{\text{mod}} &\sim \mathcal{N}(0.01,\ 0.02) \\ \bar\sigma^{\text{mod}}_\mu &\sim \mathcal{N}^{+}(0.04,\ 0.03) \\ \bar\mu^{\text{mod}}_{\log\sigma} &\sim \mathcal{N}(-1.37,\ 0.10) \\ \bar\sigma^{\text{mod}}_{\log\sigma} &\sim \mathcal{N}^{+}(0.34,\ 0.10) \\[6pt] % Hyperpriors — med (weakly informative) \bar\mu^{\text{med}},\ \bar\mu^{\text{med}}_{\log\sigma} &\sim \mathcal{N}(0, 1) \\ \bar\sigma^{\text{med}}_\mu,\ \bar\sigma^{\text{med}}_{\log\sigma} &\sim \mathcal{N}^{+}(0, 1) \\[4pt] % Indexing n &= 1, \ldots, N, \quad j[n] \in \{1, \ldots, J\} \end{align*} \]

The hyperpriors we’re using here are directly obtained from model.x.normal from t2-HighestPoint.qmd.
