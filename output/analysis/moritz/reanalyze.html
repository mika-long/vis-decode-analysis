Perform the correction:

Code 
exp2 %>%
  group_by(assignmentid, mark) %>%
  mutate(subj_mean = mean(normalized_error)) %>%
  ungroup() %>%
  mutate(grand_mean = mean(normalized_error)) %>%
  mutate(corrected = normalized_error - subj_mean + grand_mean) %>%
  group_by(mark, variability) %>%
  summarise(
    mu = mean(normalized_error),
    se = sd(corrected) / sqrt(n()) * sqrt(2 / (2 - 1)), # k is 2 when the within-subj variability is this
    .groups = "drop"
  )

If we do it the same way by calling functions in Functions.R this is what we get:

Analyzed signed error

Instead of calculating corrected error, we will first calculate error that still uses the “wrong” mean, but is not “normalized”:

Code 
freq_summary <- exp2 %>%
  group_by(assignmentid, mark) %>%
  mutate(subj_mean = mean(error)) %>%
  ungroup() %>%
  mutate(grand_mean = mean(error)) %>%
  mutate(corrected = error - subj_mean + grand_mean) %>%
  group_by(mark, variability, flip) %>%
  summarise(
    mu = mean(error),
    se = sd(corrected) / sqrt(n()) * sqrt(4 / (4 - 1)), # <- NEW: k is 4
    .groups = "drop"
  )

Recalculate error

It is unclear to me how error is calculated in the original dataset.

exp2 <- rbind(exp2_dotArc, exp2_dotX) %>%
  as_tibble(.)

# normalized error is just error but flipped when flip is TRUE
# the following confirms it.
exp2 %>% count(mark, flip, error == normalized_error)
# and error is just reportedaverage - average
# confirmed by the following
exp2 %>%
  mutate(
    error.1 = reportedaverage - data_average,
    error.2 = reportedaverage - average,
    error.3 = reportedaverage - average_along_arc,
  ) %>%
  select(seed, error.1, error.2, error.3, error, normalized_error) %>%
  arrange(seed) %>%
  filter(!near(error.2, error)) # using dplyr::near becasue of floating point related stuff ....

So let’s calculate the true error and plot things:

  unnest(col = data)

ans_df <- stimuli_df %>%
