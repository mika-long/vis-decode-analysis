# --- share helper functions
# return visual angle in degrees and not radian
vis_angle <- function(size, distance) {
  return(2 * atan(size / (2 * distance)) * 180 / pi)
}

# define function for turning vis angle to phy size
phys_size <- function(va, distance) {
  return(2 * distance * tan(va * pi / 360))
}

px_to_phys = \(px, px_per_mm) px / px_per_mm
phys_to_px = \(phys, px_per_mm) phys * px_per_mm

# --- Factory function
make_va_scale <- function(data_min, data_max, px_min, px_max) {
  px_extent <- abs(px_max - px_min)
  data_range <- data_max - data_min

  data_to_phys <- function(data, px_per_mm) {
    (data - data_min) / data_range * px_extent / px_per_mm
  }
  phys_to_data <- function(phys, px_per_mm) {
    phys * px_per_mm / px_extent * data_range + data_min
  }

  data_to_va <- function(data, distance, px_per_mm) {
    data |> data_to_phys(px_per_mm) |> vis_angle(distance = distance)
  }
  va_to_data <- function(va, distance, px_per_mm) {
    va |> phys_size(distance = distance) |> phys_to_data(px_per_mm)
  }

  data_to_px <- function(data) {
    (data - data_min) / data_range * (px_max - px_min) + px_min
  }
  px_to_data <- function(px) {
    (px - px_min) / (px_max - px_min) * data_range + data_min
  }

  list(
    data_to_va = data_to_va,
    va_to_data = va_to_data,
    data_to_phys = data_to_phys,
    phys_to_data = phys_to_data,
    data_to_px = data_to_px,
    px_to_data = px_to_data
  )
}

Model Specification

Recall the we defined the following model for the “project from dot to axis” operator:

\[ \begin{align*} \texttt{error}_i &\sim \mathcal{N}(\beta_0 + u_{\text{PID[i]}}, \sigma_{i, \text{PID}[i]}) \\ \log(\sigma_{i, j}) &= \alpha_j + \log(\texttt{va.ans.x}_i) \\ \text{or:} \log(\sigma_{i, j}) &\sim \mathcal{N}(\mu_\alpha + \log(\texttt{va.ans.x}_i), \sigma_\alpha)\\ \alpha_j &\sim N(\mu_\alpha, \sigma_\alpha) \\ \mu_\alpha &\sim N(0, 1) \\ \sigma_\alpha &\sim \mathcal{N}^+(0, 1) \end{align*} \]

Here is the specification we used for fitting:

# Recall that we defined error = resp - ans
f <- bf(
  va.error.y ~ (1 | participantId),
  sigma ~ (1 | participantId) + offset(log(va.ans.x))
)

p <- c(
  prior(normal(0, 1), class = Intercept),
  prior(normal(0, 1), class = sd, group = participantId, lb = 0),
  prior(normal(0, 1), class = Intercept, dpar = "sigma"),
  prior(normal(0, 1), class = sd, group = participantId, dpar = sigma, lb = 0)
)

To test whether our model is sensitive to prior specifications, we fit them on differnt priors:

# Wider — less informative
p_wide <- c(
  prior(normal(0, 5), class = Intercept),
  prior(normal(0, 5), class = sd, group = participantId, lb = 0),
  prior(normal(0, 5), class = Intercept, dpar = "sigma"),
  prior(normal(0, 5), class = sd, group = participantId, dpar = sigma, lb = 0)
)

# Tighter — more informative
p_tight <- c(
  prior(normal(0, 0.1), class = Intercept),
  prior(normal(0, 0.1), class = sd, group = participantId, lb = 0),
  prior(normal(0, 0.1), class = Intercept, dpar = "sigma"),
  prior(normal(0, 0.1), class = sd, group = participantId, dpar = sigma, lb = 0)
)
